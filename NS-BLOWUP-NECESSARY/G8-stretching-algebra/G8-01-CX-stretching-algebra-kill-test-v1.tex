\section{Frozen input}
\begin{center}
$\boxed{G7=\mathrm{FROZEN}}$\\
$\boxed{\mathcal C_{\mathrm{PROVEN}}=\{C_E^{weak},C_{L^3},C_F^{conc},C_I^{freq},C_J^{neg}\}}$\\
$\boxed{T^*<\infty\Rightarrow\int_0^{T^*}W_+=+\infty,\ W_+=\left(\int\omega^TS\omega\right)_+}$
\end{center}

\section{Mission G8-01}
\begin{center}
$\boxed{\mathcal C_{\mathrm{PROVEN}}\stackrel{?}{\Rightarrow}\int_0^{T^*}W_+<\infty\ \text{or}\ \bot}$\\
Target A: $W_+\le F(X_k)$, $\int_0^{T^*}F<\infty$ $\Rightarrow$ OPEN$\to$PROVEN\\
Target B: $\mathcal C_{\mathrm{PROVEN}}\Rightarrow\bot$\\
Target C: KILLED if only dimensional estimate
\end{center}

\section{Object}
$W=\int_{\mathbb R^3}\omega_i S_{ij}\omega_j\,dx$, $S=\tfrac12(\nabla u+\nabla u^T)$, $\omega=\nabla\times u$.
Decomposition: $\omega=|\omega|\xi$, $W=\int|\omega|^2\alpha$, $\alpha=\xi^TS\xi$.

\section{Barriers imposed at t=0}
\begin{itemize}
\item No $\|S\|_{L^\infty}<\infty$ without proof.
\item No $|\omega^TS\omega|\le|\omega|^2\|S\|$ as bridge to integrability.
\item No $C_{L^3}:\|u\|_{L^3}\to\infty$ $\Rightarrow$ claim on $W_+$.
\item No dimensional estimate counts: $\boxed{\text{no }|W|\lesssim\|u\|_3\|\nabla\omega\|_2^2\text{ as bridge}}$.
\end{itemize}

\section{Attempt at algebraic bridge}
Try orthogonal decomposition $S=S^+ - S^-$, $S^\pm\ge0$.
Then $W=W^+-W^-$, $W^+=\int\omega^TS^+\omega$.
$W_+\le W^+$ but $W^+\le\int|\omega|^2|S|$.

Paraproduct: $\omega_iS_{ij}\omega_j\sim\sum_{N}P_N\omega\cdot S_{\sim N}\cdot P_{\sim N}\omega$.

Attempt $W_+\lesssim\|S\|_{BMO}\|\omega\|_2^2$ or $\|S\|_{L^3}\|\omega\|_{L^3}^2$ or $\|u\|_{BMO^{-1}}\|\nabla\omega\|_2^2$.
All reduce to known: $\|S\|_{L^3}\sim\|\nabla u\|_{L^3}$, $\|\omega\|_{L^3}\sim\|\nabla u\|_{L^3}$.

Hence $\boxed{W_+\lesssim\|u\|_{L^3}\|\nabla\omega\|_2^2\text{ up to constants}}$ inevitable without new structure.

\section{Test against $\mathcal C_{\mathrm{PROVEN}}$}
$C_{L^3}$ allows $\|u(t)\|_{L^3}\to\infty$. $C_F^{conc}+C_I^{freq}$ allow $N_c(t)\to\infty$ so $\|\nabla\omega\|_2$ not controlled.
Thus $F(t)=\|u\|_3\|\nabla\omega\|_2^2$ satisfies $\int_0^{T^*}F=+\infty$ compatible with $\mathcal C_{\mathrm{PROVEN}}$.

No algebraic identity $S=\nabla_{\text{sym}}u$ gives cancellation for $(\omega^TS\omega)_+$ beyond trace-free property $\operatorname{tr}S=0$.
Trace-free gives $\lambda_1+\lambda_2+\lambda_3=0$, but does not bound positive part: can have $\lambda_1\gg1$, $\lambda_2+\lambda_3=-\lambda_1$, $\alpha_+$ still large if $\xi$ aligns with $\lambda_1$ eigenvector. $C_J^{neg}$ says alignment can be lost, but not quantified into integrable bound.

\section{Outcome}
\begin{center}
$\boxed{\mathrm{KILLED}}$\\
No $W_+\le F$ with $\int F<\infty$ from pure algebra.\\
Ends in dimensional estimate $\|u\|_3\|\nabla\omega\|_2^2$ which is allowed to diverge under $\mathcal C_{\mathrm{PROVEN}}$.\\
$\mathcal C_{\mathrm{PROVEN}}\not\Rightarrow\bot$ remains.
\end{center}
KILL conditions met: (i) only $|\omega|^2\|S\|$, (ii) needs $\|S\|_\infty$, (iii) $F$ not integrable, (iv) no new structure beyond trace-free.
