{\Large NS-BLOWUP-F-04: VEREDICTO log-subcritico}\\
Juicio final $B^{-1}_{\infty,1}$ vs $B^{-1}_{\infty,\infty}$
\end{center}

\paragraph{Mision F-04:} decidir si etiqueta log-subcritico tiene fundamento exacto.

\subsection*{1. Escala NS}
\[
u_\lambda(x,t)=\lambda u(\lambda x,\lambda^2 t)
\]
\[
\boxed{\|u_\lambda\|_{B^{-1}_{\infty,q}}=\|u\|_{B^{-1}_{\infty,q}},\ q=1,\infty}
\]
\[
\boxed{F1:\text{ CRITICO para }q=1,\infty}
\]

\subsection*{2. Que aporta $q=1$?}
\[
\boxed{B^{-1}_{\infty,1}\subset B^{-1}_{\infty,\infty}}
\]
\[
\sum_j 2^{-j}\|\Delta_j u\|_\infty
\]
\[
\boxed{\text{suma fina genera mecanismo cuantitativo?}}
\]

\subsection*{3. Mecanismo entra en PDE?}
\[
\boxed{B^{-1}_{\infty,1}\to\text{estim NS}\to\text{control blow-up}}
\]
Si no puente:
\[
\boxed{F2=OPEN}
\]

\subsection*{4. Veredicto}
VERDE RESCATE:
\[
\boxed{B^{-1}_{\infty,1}\to\text{control NS}\to T^*=\infty\Rightarrow JUSTIFICADO}
\]
ROJO MUERTE:
\[
\boxed{\text{ solo inclusion sin puente NS}\Rightarrow\text{NO JUSTIFICADO}}
\]
\[
\boxed{\text{Eliminar vocabulario oficial F}}
\]

\subsection*{Regla ZFL}
\[
\boxed{F\text{-}04=\text{falsacion ruta}=resultado}
\]
\[
\boxed{T^*<\infty\Rightarrow C_E,\quad T^*<\infty\Rightarrow C_F,\quad C_E\land C_F\Rightarrow\bot}
\]

\subsection*{Estado}
\[
\boxed{F\text{-}04: OPEN - juicio JUSTIFICADO o MUERTE}
\]
